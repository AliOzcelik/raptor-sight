#
#   sudo apt install python3-picamera2 python3-opencv python3-numpy python3-yaml python3-psutil

# Hailo-10H runtime and PCIe driver (version 5.4.0, the version the HEF files are compiled for):
#
#   sudo apt install ./hailort_5.4.0_arm64.deb ./hailort-pcie-driver_5.4.0_all.deb
#
# Create the venv so it can see the apt packages:
#
#   python3 -m venv --system-site-packages .venv
#
# ---------------------------------------------------------------------------------------
# 2. HailoRT Python API (pip, from Hailo's wheel, not on PyPI)
# ---------------------------------------------------------------------------------------
# The apt package python3-h10-hailort is 5.1.1 and doesn't match the 5.4.0 runtime, so use
# the 5.4.0 wheel from Hailo's Developer Zone:
#
#   .venv/bin/pip install ~/Downloads/hailort-5.4.0-cp313-cp313-linux_aarch64.whl
#
# ---------------------------------------------------------------------------------------
# 3. pip requirements (perception and benchmarks: what's in use now)
# ---------------------------------------------------------------------------------------

# them as already installed and changes nothing.
